CREATE PROCEDURE sp_ObtenerPacienteId
    @id_usuario INT,
    @id_paciente INT OUTPUT,
    @mensaje NVARCHAR(255) OUTPUT
AS
BEGIN
    SELECT @id_paciente = id_paciente 
    FROM paciente 
    WHERE id_usuario = @id_usuario;
    
    IF @id_paciente IS NULL
        SET @mensaje = 'No eres paciente. Debes registrarte como paciente primero.';
    ELSE
        SET @mensaje = 'OK';
END

go
CREATE PROCEDURE sp_VerificarInscripcionExistente
    @id_paciente INT,
    @id_turno INT,
    @existe BIT OUTPUT,
    @mensaje NVARCHAR(255) OUTPUT
AS
BEGIN
    IF EXISTS (
        SELECT 1 FROM inscripcion 
        WHERE id_paciente = @id_paciente AND id_turno = @id_turno
    )
    BEGIN
        SET @existe = 1;
        SET @mensaje = 'Ya estás inscrito en este turno';
    END
    ELSE
    BEGIN
        SET @existe = 0;
        SET @mensaje = 'OK';
    END
END

go
CREATE PROCEDURE sp_CrearInscripcion
    @id_paciente INT,
    @id_turno INT,
    @id_inscripcion INT OUTPUT,
    @mensaje NVARCHAR(255) OUTPUT
AS
BEGIN
    BEGIN TRANSACTION;
    
    BEGIN TRY
        INSERT INTO inscripcion (id_turno, id_paciente, fecha_inscripcion)
        VALUES (@id_turno, @id_paciente, GETDATE());
        
        SET @id_inscripcion = SCOPE_IDENTITY();
        
        UPDATE turno SET estado = 'ocupado' WHERE id_turno = @id_turno;
        
        SET @mensaje = 'Inscripción exitosa';
        
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SET @id_inscripcion = NULL;
        SET @mensaje = ERROR_MESSAGE();
    END CATCH
END
go
-- ============================================
-- 4. sp_GetInscripcionesByPaciente
-- ============================================
CREATE PROCEDURE sp_GetInscripcionesByPaciente
    @id_paciente INT
AS
BEGIN
    SELECT 
        i.id_inscripcion,
        i.fecha_inscripcion,
        t.id_turno,
        t.fecha AS fecha_turno,
        t.hora_inicio,
        t.hora_fin,
        t.estado AS estado_turno,
        m.especialidad,
        u.nombre AS medico_nombre,
        u.email AS medico_email,
        u.telefono AS medico_telefono
    FROM inscripcion i
    INNER JOIN turno t ON i.id_turno = t.id_turno
    INNER JOIN medico m ON t.id_medico = m.id_medico
    INNER JOIN usuario u ON m.id_usuario = u.id_usuario
    WHERE i.id_paciente = @id_paciente
    ORDER BY i.fecha_inscripcion DESC
END
go
-- ============================================
-- 5. sp_VerificarPropiedadInscripcion
-- ============================================
CREATE PROCEDURE sp_VerificarPropiedadInscripcion
    @id_inscripcion INT,
    @id_paciente INT,
    @pertenece BIT OUTPUT,
    @mensaje NVARCHAR(255) OUTPUT
AS
BEGIN
    IF EXISTS (
        SELECT 1 FROM inscripcion 
        WHERE id_inscripcion = @id_inscripcion AND id_paciente = @id_paciente
    )
    BEGIN
        SET @pertenece = 1;
        SET @mensaje = 'OK';
    END
    ELSE
    BEGIN
        SET @pertenece = 0;
        SET @mensaje = 'Esta inscripción no te pertenece';
    END
END
go
-- ============================================
-- 6. sp_CancelarInscripcion
-- ============================================
CREATE PROCEDURE sp_CancelarInscripcion
    @id_inscripcion INT
AS
BEGIN
    DECLARE @id_turno INT;
    
    SELECT @id_turno = id_turno FROM inscripcion WHERE id_inscripcion = @id_inscripcion;
    
    DELETE FROM inscripcion WHERE id_inscripcion = @id_inscripcion;
    
    UPDATE turno SET estado = 'disponible' WHERE id_turno = @id_turno;
END
